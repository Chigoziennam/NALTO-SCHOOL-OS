# Everything a crawler can see, it may index. The shop is one page, so there is
# no section to hold back. The one exception is /api/: those endpoints take
# money and read Paystack, they return nothing a searcher wants, and a crawler
# hitting them repeatedly is pure noise in the logs.
#
# A plain-language summary of the brand, the full catalogue and the prices lives
# at /llms.txt, for anything that would rather read prose than parse the page.

User-agent: *
Allow: /
Disallow: /api/

# The product and library photos are the point of a clothing brand in image
# search, so they stay open to the image crawlers by name as well.
User-agent: Googlebot-Image
Allow: /img/

User-agent: Bingbot
Allow: /
Disallow: /api/
